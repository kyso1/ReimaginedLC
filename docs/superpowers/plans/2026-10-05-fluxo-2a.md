# Subprojeto 2a: plano de implementação

> **Para quem executa:** este plano é executado no modo nativo (`superpowers:executing-plans`), pela sessão principal, sem subagente por tarefa. Os passos usam checkbox (`- [ ]`).

**Objetivo:** controles de janela, convites, tela de modos, lobby Palco, sala personalizada, fila e aceite de partida dentro da camada do ReimaginedLC.

**Arquitetura:** estende a fundação. Um `createStore` por endpoint novo, uma função `request` para ações, o roteador ganha `lobby`, e uma camada de sobreposições no shell fica acima da tela atual. Lógica pura (agrupamento de filas, corpo da sala, visibilidade do aceite, rotas) em funções testadas; telas em Preact com um teste por tela contra fixtures.

**Stack:** Vite, Preact, `@preact/signals`, TypeScript, Vitest com jsdom. Build `chrome108`.

**Spec:** `docs/superpowers/specs/2026-10-05-fluxo-2a-design.md`

**Desvio consciente do formato padrão do writing-plans:** o dono pediu execução enxuta e quem executa é a mesma sessão que escreveu a spec. O plano traz o código completo da lógica pura, dos stores e dos testes que fixam contratos, e para as telas traz estrutura, interfaces, textos e os testes exigidos, sem transcrever cada linha de JSX.

```ts
// src/lcu/request.ts
import { t } from '../i18n';

export class LcuError extends Error {}

type Method = 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE';

/** Ação na LCU. Em erro mostra a mensagem da LCU num Toast e rejeita; quem chama não muda estado local. */
export async function request<T = unknown>(method: Method, path: string, body?: unknown, fetcher: typeof fetch = (u, i) => fetch(u, i)): Promise<T | null> {
  const init: RequestInit = body === undefined ? { method } : { method, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) };
  let response: Response;
  try {
    response = await fetcher(path, init);
  } catch (error) {
    window.Toast?.error(t('lcuOffline'));
    throw new LcuError(String(error));
  }
  const text = await response.text().catch(() => '');
  if (!response.ok) {
    const message = lcuMessage(text) ?? `${t('lcuError')} (${response.status})`;
    window.Toast?.error(message);
    throw new LcuError(message);
  }
  return text ? (JSON.parse(text) as T) : null;
}

function lcuMessage(text: string): string | null {
  try {
    const parsed = JSON.parse(text) as { message?: unknown };
    return typeof parsed?.message === 'string' && parsed.message ? parsed.message : null;
  } catch {
    return null;
  }
}
```

Textos: `lcuError: 'Erro da LCU' / 'LCU error'`, `lcuOffline: 'Sem conexão com o client' / 'No connection to the client'`.

```ts
// src/ui/useAction.ts
import { useRef, useState } from 'preact/hooks';

/** Roda uma ação da LCU uma vez por vez. A rejeição já foi mostrada por `request`. */
export function useAction<A extends unknown[]>(fn: (...args: A) => Promise<unknown>) {
  const [pending, setPending] = useState(false);
  const busy = useRef(false);
  const run = (...args: A) => {
    if (busy.current) return;
    busy.current = true;
    setPending(true);
    fn(...args).catch(() => {}).finally(() => {
      busy.current = false;
      setPending(false);
    });
  };
  return { run, pending };
}
```

```ts
import type { LolGameQueuesQueue, LolLobbyLobbyChangeGameDto } from '../../lcu/types';

export type ModeGroupId = 'sr' | 'aram' | 'tft' | 'other';
export interface ModeGroup { id: ModeGroupId; queues: LolGameQueuesQueue[] }
export interface CustomForm { name: string; password: string; teamSize: number; spectators: 'AllAllowed' | 'LobbyAllowed' | 'NotAllowed' }

const GROUP: Record<string, ModeGroupId> = { kSummonersRift: 'sr', kARAM: 'aram', kTeamfightTactics: 'tft' };
const ORDER: ModeGroupId[] = ['sr', 'aram', 'tft', 'other'];
const usable = (q: LolGameQueuesQueue) => q.queueAvailability === 'Available' && q.isVisible !== false;

export function groupQueues(queues: LolGameQueuesQueue[]): ModeGroup[] {
  const buckets = new Map<ModeGroupId, LolGameQueuesQueue[]>();
  for (const q of queues) {
    if (!usable(q) || q.gameSelectCategory === 'CreateCustom') continue;
    const id = GROUP[q.gameSelectModeGroup] ?? 'other';
    buckets.set(id, [...(buckets.get(id) ?? []), q]);
  }
  return ORDER.filter((id) => buckets.has(id)).map((id) => ({
    id,
    queues: buckets.get(id)!.map((q, i) => ({ q, i })).sort((a, b) => (b.q.gameSelectPriority ?? 0) - (a.q.gameSelectPriority ?? 0) || a.i - b.i).map(({ q }) => q),
  }));
}

export const customModes = (queues: LolGameQueuesQueue[]) =>
  queues.filter((q) => usable(q) && q.gameSelectCategory === 'CreateCustom').sort((a, b) => a.name.localeCompare(b.name));

export function customLobbyBody(mode: LolGameQueuesQueue, form: CustomForm): LolLobbyLobbyChangeGameDto & { isCustom: true } {
  return {
    queueId: mode.id,
    isCustom: true,
    customGameLobby: {
      lobbyName: form.name.trim(),
      lobbyPassword: form.password,
      configuration: { gameMode: mode.gameMode, mapId: mode.mapId, mutators: { id: mode.gameTypeConfig.id }, spectatorPolicy: form.spectators, teamSize: Math.min(5, Math.max(1, Math.round(form.teamSize))) },
    },
  } as LolLobbyLobbyChangeGameDto & { isCustom: true };
}
```
