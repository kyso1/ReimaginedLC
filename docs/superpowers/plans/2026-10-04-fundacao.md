**Goal:** Entregar o esqueleto do plugin ReimaginedLC: carrega no client, cobre a janela com a nossa camada, mostra a tela inicial no tema Vidro, troca de tema ao vivo e sai da frente com `Ctrl+Shift+F12`.

**Tech Stack:** Vite 5, Preact 10, `@preact/signals`, TypeScript 5, Vitest com jsdom para `src/`, `node:test` para `tools/`, pacote `ws` (já instalado) na LCU simulada.

## Desvios deliberados da spec

Dois detalhes da spec não funcionam como escritos; o plano os resolve assim, e a Task 8 registra a mudança na spec:

1. **Pasta dos temas embutidos.** A spec diz `themes/` na raiz. O Vite só copia para o bundle o que está em `public/`, então os temas ficam em `public/themes/` e saem em `dist/ReimaginedLC/themes/`.
2. **Mídia dos temas embutidos.** A spec só aceita caminho relativo à pasta do tema. Como não podemos commitar mídia da Riot, o validador aceita também URLs de asset do próprio client, começando por `/fe/` ou `/lol-game-data/assets/`. Valor vazio (`""`) significa "sem mídia" e serve para um tema filho anular o fundo herdado.

## Review Focus

1. **Tema malformado ou hostil.** Um `theme.json` com valor de token que não é do formato esperado (por exemplo `red; background: url(http://x)`), caminho de mídia com `..` ou com esquema, `extends` em ciclo: o tema é recusado inteiro, o Vidro é aplicado e a mensagem diz o campo. Testes na Task 3.
2. **Valor inesperado no `DataStore`.** `rlc.enabled` que não é booleano, `rlc.theme` que não é objeto ou aponta para tema inexistente: a camada fica desligada e o tema cai no Vidro, sem exceção. Testes nas Tasks 1 e 4.
3. **Socket da LCU cai.** Reconexão com espera crescente e novo `GET` de cada store. Evento com JSON inválido ou formato inesperado é ignorado. Testes na Task 2.
4. **Tela lança exceção ao renderizar.** A camada se desliga e o client original aparece. Teste na Task 6.
5. **Fase do gameflow sem tela nossa ou store indisponível.** A camada se oculta; nunca fica um root vazio cobrindo o client. Teste na Task 6.

```ts
// Configuração do plugin no DataStore do Pengu. No modo navegador o DataStore pode não existir.
export function readSetting<T>(key: string, fallback: T): T {
  try {
    const store = window.DataStore;
    if (!store || !store.has(key)) return fallback;
    return store.get(key) as T;
  } catch {
    return fallback;
  }
}

export function writeSetting(key: string, value: unknown): void {
  try {
    window.DataStore?.set(key, value);
  } catch {
    // Sem DataStore utilizável o estado vale só para esta sessão.
  }
}
```

```ts
import { signal } from '@preact/signals';
import { t } from '../i18n';
import { readSetting, writeSetting } from './storage';

/** Camada do ReimaginedLC visível. Desligada, o client original aparece e funciona normalmente. */
export const enabled = signal(false);
/** Com a camada ligada, esconder a interface da Riot (`visibility: hidden`) em vez de só cobri-la. */
export const hideRiot = signal(true);

/** Relê a configuração. Chamado em `load()`, quando o DataStore já existe. */
export function syncLayer(): void {
  enabled.value = readSetting<unknown>('rlc.enabled', false) === true;
  hideRiot.value = readSetting<unknown>('rlc.hideRiot', true) !== false;
}

export function setEnabled(value: boolean): void {
  enabled.value = value;
  writeSetting('rlc.enabled', value);
  window.Toast?.success(value ? t('layerOn') : t('layerOff'));
}

export const toggleEnabled = (): void => setEnabled(!enabled.value);
```

```ts
import { toggleEnabled } from './layer';

let registered = false;

/** Ctrl+Shift+F12 alterna a camada. Registrado em `init`, antes de qualquer outro código do plugin. */
export function registerHotkey(target: Window = window): void {
  if (registered) return;
  registered = true;
  target.addEventListener(
    'keydown',
    (e) => {
      if (!(e.ctrlKey && e.shiftKey && e.key === 'F12')) return;
      e.preventDefault();
      e.stopPropagation();
      toggleEnabled();
    },
    true,
  );
}
```

```ts
type Listener = (eventType: string, data: unknown) => void;

export interface LcuSocket {
  subscribe(uri: string, fn: Listener): () => void;
  onReconnect(fn: () => void): () => void;
  close(): void;
}

/** Espera antes da tentativa `attempt` de reconexão: 1 s, 2 s, 4 s... até 30 s. */
export const backoff = (attempt: number): number => Math.min(30000, 1000 * 2 ** attempt);

export function connect(
  url: string,
  WS: typeof WebSocket = WebSocket,
  wait: (fn: () => void, ms: number) => unknown = (fn, ms) => setTimeout(fn, ms),
): LcuSocket {
  const listeners = new Map<string, Set<Listener>>();
  const reconnectListeners = new Set<() => void>();
  let ws: WebSocket | null = null;
  let attempt = 0;
  let everOpened = false;
  let closed = false;

  const open = () => {
    const socket = new WS(url, 'wamp');
    ws = socket;
    socket.onopen = () => {
      socket.send(JSON.stringify([5, 'OnJsonApiEvent']));
      attempt = 0;
      if (everOpened) reconnectListeners.forEach((fn) => fn());
      everOpened = true;
    };
    socket.onmessage = (message) => {
      let parsed: unknown;
      try {
        parsed = JSON.parse(String(message.data));
      } catch {
        return;
      }
      if (!Array.isArray(parsed) || parsed[0] !== 8) return;
      const payload = parsed[2] as { uri?: unknown; eventType?: unknown; data?: unknown } | undefined;
      if (!payload || typeof payload.uri !== 'string') return;
      listeners.get(payload.uri)?.forEach((fn) => fn(String(payload.eventType), payload.data));
    };
    socket.onclose = () => {
      if (closed) return;
      wait(open, backoff(attempt++));
    };
  };
  open();

  return {
    subscribe(uri, fn) {
      const set = listeners.get(uri) ?? new Set<Listener>();
      listeners.set(uri, set);
      set.add(fn);
      return () => void set.delete(fn);
    },
    onReconnect(fn) {
      reconnectListeners.add(fn);
      return () => void reconnectListeners.delete(fn);
    },
    close() {
      closed = true;
      ws?.close();
    },
  };
}
```

```ts
import { signal, type Signal } from '@preact/signals';
import type { LcuSocket } from './socket';

export type StoreState<T> = { status: 'loading' | 'ready' | 'unavailable'; data: T | null };

/**
 * Estado de um recurso da LCU: um GET inicial e depois os eventos daquela URI.
 * 404 no GET é estado válido (o recurso não existe agora), não erro.
 */
export function createStore<T>(
  uri: string,
  socket: LcuSocket,
  get: (uri: string) => Promise<Response> = (u) => fetch(u),
): Signal<StoreState<T>> {
  const state = signal<StoreState<T>>({ status: 'loading', data: null });

  // ponytail: um evento que chegue antes de o GET terminar é sobrescrito pela resposta do GET.
  // A janela é de milissegundos e o próximo evento corrige; se virar problema, numerar as atualizações.
  const refresh = async () => {
    try {
      const response = await get(uri);
      if (response.status === 404) state.value = { status: 'ready', data: null };
      else if (!response.ok) state.value = { status: 'unavailable', data: null };
      else state.value = { status: 'ready', data: (await response.json()) as T };
    } catch {
      state.value = { status: 'unavailable', data: null };
    }
  };

  socket.subscribe(uri, (eventType, data) => {
    state.value = { status: 'ready', data: eventType === 'Delete' ? null : (data as T) };
  });
  socket.onReconnect(() => void refresh());
  void refresh();
  return state;
}
```

```ts
  'radius.sm': 'length',
  'radius.md': 'length',
  'radius.pill': 'length',
  'blur.surface': 'length',
  'motion.duration': 'duration',
  'motion.easing': 'easing',
};

export const TRANSITIONS: readonly Transition[] = ['wipe', 'fade', 'slide', 'zoom'];

// Um tema é só dado. Cada valor precisa casar inteiro com o formato do seu tipo;
// nada com `;`, `{`, `url(` ou `var(` passa, porque nenhum formato abaixo os admite.
const FORMAT: Record<TokenKind, RegExp> = {
  color: /^(#[0-9a-f]{3,8}|(rgb|rgba|hsl|hsla)\([\d\s.,%/]+\))$/i,
  length: /^\d+(\.\d+)?(px|rem|%)$/,
  duration: /^\d+(\.\d+)?(ms|s)$/,
  weight: /^[1-9]\d0$/,
  easing: /^(linear|ease|ease-in|ease-out|ease-in-out|cubic-bezier\(\s*-?\d*\.?\d+\s*(,\s*-?\d*\.?\d+\s*){3}\))$/,
  font: /^[\p{L}\p{N} ,'"_-]+$/u,
};

const NAME = /^[a-z0-9-]{1,40}$/;
const SIMPLE = /^[a-z0-9-]{1,40}$/i;
const MEDIA_EXT = '(png|jpg|webp|gif|webm|mp4)';
const RELATIVE_MEDIA = new RegExp(`^[\\w.-]+(/[\\w.-]+)*\\.${MEDIA_EXT}$`, 'i');
const CLIENT_MEDIA = new RegExp(`^/(fe|lol-game-data/assets)(/[\\w.-]+)+\\.${MEDIA_EXT}$`, 'i');
const ROOT_KEYS = new Set(['name', 'label', 'version', 'extends', 'tokens', 'transition', 'backgrounds', 'layouts']);

const isRecord = (v: unknown): v is Record<string, unknown> => typeof v === 'object' && v !== null && !Array.isArray(v);
const validMedia = (path: string): boolean =>
  path === '' || (!path.includes('..') && (RELATIVE_MEDIA.test(path) || CLIENT_MEDIA.test(path)));

type Result = { ok: true; theme: ThemeManifest } | { ok: false; error: string };
const fail = (field: string, why: string): Result => ({ ok: false, error: `${field}: ${why}` });

export function validateTheme(input: unknown): Result {
  if (!isRecord(input)) return fail('tema', 'precisa ser um objeto');
  for (const key of Object.keys(input)) if (!ROOT_KEYS.has(key)) return fail(key, 'campo desconhecido');

  const { name, label, version, tokens = {}, transition, backgrounds = {}, layouts = {} } = input;
  const parent = input.extends ?? null;

  if (typeof name !== 'string' || !NAME.test(name)) return fail('name', 'use só minúsculas, dígitos e hífen (até 40)');
  if (typeof label !== 'string' || label.length === 0 || label.length > 60) return fail('label', 'texto de 1 a 60 caracteres');
  if (version !== 1) return fail('version', 'só a versão 1 é suportada');
  if (parent !== null && (typeof parent !== 'string' || !NAME.test(parent))) return fail('extends', 'nome de tema inválido');
  if (transition !== undefined && !TRANSITIONS.includes(transition as Transition)) return fail('transition', `use um de ${TRANSITIONS.join(', ')}`);

  if (!isRecord(tokens)) return fail('tokens', 'precisa ser um objeto');
  for (const [token, value] of Object.entries(tokens)) {
    const kind = TOKEN_KINDS[token];
    if (!kind) return fail(`tokens.${token}`, 'token desconhecido');
    if (typeof value !== 'string' || !FORMAT[kind].test(value)) return fail(`tokens.${token}`, `valor fora do formato de ${kind}`);
    if (kind === 'weight' && Number(value) > 900) return fail(`tokens.${token}`, 'peso de 100 a 900');
  }

  if (!isRecord(backgrounds)) return fail('backgrounds', 'precisa ser um objeto');
  for (const [screen, path] of Object.entries(backgrounds)) {
    if (!SIMPLE.test(screen)) return fail(`backgrounds.${screen}`, 'nome de tela inválido');
    if (typeof path !== 'string' || !validMedia(path)) return fail(`backgrounds.${screen}`, 'caminho de mídia não permitido');
  }

  if (!isRecord(layouts)) return fail('layouts', 'precisa ser um objeto');
  for (const [screen, layout] of Object.entries(layouts)) {
    if (!SIMPLE.test(screen) || typeof layout !== 'string' || !SIMPLE.test(layout)) return fail(`layouts.${screen}`, 'nome de layout inválido');
  }

  return {
    ok: true,
    theme: {
      name,
      label,
      version: 1,
      extends: parent as string | null,
      tokens: tokens as Record<string, string>,
      ...(transition ? { transition: transition as Transition } : {}),
      backgrounds: backgrounds as Record<string, string>,
      layouts: layouts as Record<string, string>,
    },
  };
}
```

```ts
import { TOKEN_KINDS, type ThemeManifest, type Transition } from './validate';

export interface ResolvedTheme {
  name: string;
  label: string;
  tokens: Record<string, string>;
  transition: Transition;
  /** Cada fundo lembra o tema que o declarou, porque o caminho é relativo à pasta desse tema. */
  backgrounds: Record<string, { theme: string; path: string }>;
}

const MAX_DEPTH = 3;

type Result = { ok: true; theme: ResolvedTheme } | { ok: false; error: string };

export function resolveTheme(name: string, themes: Map<string, ThemeManifest>): Result {
  const chain: ThemeManifest[] = [];
  const seen = new Set<string>();
  for (let current: string | null = name; current !== null; ) {
    if (seen.has(current)) return { ok: false, error: `extends: ciclo em "${current}"` };
    seen.add(current);
    const manifest = themes.get(current);
    if (!manifest) return { ok: false, error: `tema "${current}" não encontrado` };
    chain.push(manifest);
    if (chain.length > MAX_DEPTH) return { ok: false, error: `extends: profundidade acima de ${MAX_DEPTH}` };
    current = manifest.extends;
  }

  const own = chain[0]!;
  const resolved: ResolvedTheme = { name: own.name, label: own.label, tokens: {}, transition: 'fade', backgrounds: {} };
  for (const manifest of chain.reverse()) {
    Object.assign(resolved.tokens, manifest.tokens);
    if (manifest.transition) resolved.transition = manifest.transition;
    for (const [screen, path] of Object.entries(manifest.backgrounds)) resolved.backgrounds[screen] = { theme: manifest.name, path };
  }

  const missing = Object.keys(TOKEN_KINDS).filter((token) => !(token in resolved.tokens));
  if (missing.length) return { ok: false, error: `tokens ausentes: ${missing.join(', ')}` };
  return { ok: true, theme: resolved };
}
```

```ts
import type { ResolvedTheme } from './resolve';

export const cssVar = (token: string): string => `--rlc-${token.replaceAll('.', '-')}`;

/** Grava os tokens como variáveis CSS no elemento e remove as do tema anterior. */
export function applyTheme(el: HTMLElement, theme: ResolvedTheme): void {
  for (const property of Array.from(el.style)) if (property.startsWith('--rlc-')) el.style.removeProperty(property);
  for (const [token, value] of Object.entries(theme.tokens)) el.style.setProperty(cssVar(token), value);
  el.dataset.theme = theme.name;
  el.dataset.transition = theme.transition;
}

/** URL da mídia de fundo de uma tela, ou null. `base` é a URL da pasta do plugin, terminada em `/`. */
export function mediaUrl(theme: ResolvedTheme, screen: string, base: string): string | null {
  const background = theme.backgrounds[screen] ?? theme.backgrounds.default;
  if (!background || background.path === '') return null;
  if (background.path.startsWith('/')) return background.path;
  return `${base}themes/${background.theme}/${background.path}`;
}
```

```ts
import { signal, type Signal } from '@preact/signals';
import { t } from '../i18n';
import { readSetting, writeSetting } from '../shell/storage';
import { resolveTheme, type ResolvedTheme } from './resolve';
import { validateTheme, type ThemeManifest } from './validate';

export const BUILTIN: readonly string[] = ['vidro', 'vidro-ludico', 'editorial'];
const FALLBACK = 'vidro';

export type Game = 'lol' | 'tft';

export interface ThemeEngine {
  active: Signal<ResolvedTheme>;
  available: Array<{ name: string; label: string }>;
  /** Troca o tema de um jogo. Neste subprojeto só o de `lol` muda o tema ativo. */
  set(name: string, game?: Game): void;
}

const isRecord = (v: unknown): v is Record<string, unknown> => typeof v === 'object' && v !== null && !Array.isArray(v);

async function loadManifest(base: string, name: string, get: (url: string) => Promise<Response>): Promise<ThemeManifest | string> {
  try {
    const response = await get(`${base}themes/${name}/theme.json`);
    if (!response.ok) return `${name}: HTTP ${response.status}`;
    const result = validateTheme(await response.json());
    if (!result.ok) return `${name}: ${result.error}`;
    if (result.theme.name !== name) return `${name}: name diferente da pasta`;
    return result.theme;
  } catch (e) {
    return `${name}: ${e instanceof Error ? e.message : 'falha ao carregar'}`;
  }
}

export async function createThemeEngine(base: string, get: (url: string) => Promise<Response> = (url) => fetch(url)): Promise<ThemeEngine> {
  const manifests = new Map<string, ThemeManifest>();
  for (const name of BUILTIN) {
    const loaded = await loadManifest(base, name, get);
    if (typeof loaded === 'string') window.Toast?.error(`${t('themeInvalid')}: ${loaded}`);
    else manifests.set(name, loaded);
  }

  const resolved = new Map<string, ResolvedTheme>();
  for (const name of manifests.keys()) {
    const result = resolveTheme(name, manifests);
    if (result.ok) resolved.set(name, result.theme);
    else window.Toast?.error(`${t('themeInvalid')}: ${name}: ${result.error}`);
  }

  const fallback = resolved.get(FALLBACK);
  if (!fallback) throw new Error(`tema embutido "${FALLBACK}" indisponível`);

  const stored = readSetting<unknown>('rlc.theme', null);
  const saved = isRecord(stored) ? stored : {};
  const wanted = typeof saved.lol === 'string' ? saved.lol : FALLBACK;
  const active = signal(resolved.get(wanted) ?? fallback);

  return {
    active,
    available: [...resolved.values()].map(({ name, label }) => ({ name, label })),
    set(name, game = 'lol') {
      const theme = resolved.get(name);
      if (!theme) {
        window.Toast?.error(`${t('themeInvalid')}: ${name}`);
        return;
      }
      const current = readSetting<unknown>('rlc.theme', null);
      writeSetting('rlc.theme', { ...(isRecord(current) ? current : {}), [game]: name });
      if (game === 'lol') active.value = theme;
    },
  };
}
```

```tsx
import { useEffect, useRef, useState } from 'preact/hooks';

export const isVideo = (src: string): boolean => /\.(webm|mp4)(\?|$)/i.test(src);

/** Fundo de tela do tema. Vídeo só toca enquanto `playing`; mídia que falha vira a cor de fundo. */
export function MediaBackground({ src, playing }: { src: string | null; playing: boolean }) {
  const [failed, setFailed] = useState<string | null>(null);
  const video = useRef<HTMLVideoElement>(null);

  useEffect(() => {
    const el = video.current;
    if (!el) return;
    if (playing) void el.play().catch(() => {});
    else el.pause();
  }, [playing, src]);

  if (!src || failed === src) return <div class="rlc-bg" />;
  const onError = () => setFailed(src);
  return isVideo(src) ? (
    <video ref={video} class="rlc-bg" src={src} muted loop playsInline onError={onError} />
  ) : (
    <img class="rlc-bg" src={src} alt="" onError={onError} />
  );
}
```

```tsx
/** "16.19.823.0722" vira "16.19". */
export function shortPatch(version: string | null | undefined): string | null {
  const match = /^(\d+)\.(\d+)\./.exec(version ?? '');
  return match ? `${match[1]}.${match[2]}` : null;
}

/** Página de notas do patch no site da Riot, no idioma do client. Locale fora do padrão `xx_YY` cai em inglês. */
export function patchNotesUrl(locale: string | null | undefined): string {
  const safe = /^[a-z]{2}_[A-Z]{2}$/.test(locale ?? '') ? locale!.toLowerCase().replace('_', '-') : 'en-us';
  return `https://www.leagueoflegends.com/${safe}/news/tags/patch-notes/`;
}

export function Home({ version, locale, onPlay, open = (url) => window.open(url, '_blank', 'noopener') }: { version: string | null; locale: string | null; onPlay: () => void; open?: (url: string) => unknown }) {

  const openNotes = () => {
    try {
      if (!open(patchNotesUrl(locale))) window.Toast?.error(t('openFailed'));
    } catch {
      window.Toast?.error(t('openFailed'));
    }
  };
  return (
    <section class="rlc-home">
      <div class="rlc-home-play">
        <Button variant="primary" onClick={onPlay}>
          {t('play')}
        </Button>
      </div>
      <div class="rlc-home-patch">

          {t('patchNotes')}
        </Button>

  );
}
```

```tsx
export class Boundary extends Component<{ onError: (error: unknown) => void; children: ComponentChildren }, { failed: boolean }> {
  state = { failed: false };
```

```tsx
import type { ThemeEngine } from '../theme/engine';
import { MediaBackground } from '../ui/MediaBackground';
import { Rail } from '../ui/Rail';
import { Boundary } from './Boundary';
import { enabled, hideRiot, setEnabled } from './layer';
import { screenFor, type Screen } from './router';

/** Elemento raiz da interface da Riot. */
export const RIOT_ROOT_ID = 'rcp-fe-viewport-root';

type HomeProps = Parameters<typeof Home>[0];

interface Props {
  root: HTMLElement;
  lcu: Lcu;
  themes: ThemeEngine;
  base: string;
  focused: Signal<boolean>;
  screens?: Record<Screen, (props: HomeProps) => JSX.Element>;
}

export function App({ root, lcu, themes, base, focused, screens = { home: Home } }: Props) {
  const screen = screenFor(lcu.gameflowPhase.value);
  const visible = enabled.value && screen !== null;
  const theme = themes.active.value;
  const hide = visible && hideRiot.value;

  useEffect(() => applyTheme(root, theme), [root, theme]);

  useEffect(() => {
    root.dataset.visible = String(visible);
    const riot = document.getElementById(RIOT_ROOT_ID);
    if (riot) riot.style.visibility = hide ? 'hidden' : '';
  }, [root, visible, hide]);

  if (!visible || screen === null) return null;

  const turnOff = () => setEnabled(false);
  const onError = (error: unknown) => {
    console.error('[rlc]', error);
    enabled.value = false;
    window.Toast?.error(t('screenFailed'));
  };
  const ScreenComponent = screens[screen];

  return (
    <div class="rlc-app" key={theme.name}>
      <MediaBackground src={mediaUrl(theme, screen, base)} playing={focused.value} />
      <Rail
        items={[
          { id: 'home', label: t('home'), active: true, onSelect: () => {} },
          { id: 'original', label: t('originalClient'), onSelect: turnOff },
        ]}
      />
      <main class="rlc-screen">
        <Boundary onError={onError}>
          <ScreenComponent version={lcu.build.value.data?.version ?? null} locale={lcu.regionLocale.value.data?.locale ?? null} onPlay={turnOff} />
        </Boundary>
      </main>
    </div>
  );
}
```

Notas para quem implementa:
- `onError` desliga a camada só nesta sessão (`enabled.value = false`, sem gravar no `DataStore`): um erro de tela não deve deixar a camada desligada para sempre.

```js
const PHASE_URI = '/lol-gameflow/v1/gameflow-phase';
const CDN = 'https://raw.communitydragon.org/latest/plugins';

const valid = (line) => line && typeof line === 'object' && typeof line.uri === 'string' && (line.type === 'snapshot' || line.type === 'event');

/** Reproduz uma gravação de sessão: estado para os GETs e a fila de eventos do WebSocket. */
export function createReplay(lines, { idle = false } = {}) {
  const state = new Map();
  const events = [];
  for (const line of lines) {
    if (!valid(line)) continue;
    if (line.type === 'snapshot') state.set(line.uri, { status: Number(line.status) || 200, data: line.data ?? null });
    else events.push(line);
  }
  if (idle) state.set(PHASE_URI, { status: 200, data: 'None' });

  let index = 0;
  let clock = 0;

  return {
    get(uri) {
      return state.get(uri.split('?')[0]) ?? { status: 404, data: { errorCode: 'RESOURCE_NOT_FOUND', httpStatus: 404, message: 'Not found in recording' } };
    },
    next() {
      if (idle || index >= events.length) return null;
      const event = events[index++];
      const t = Number(event.t) || 0;
      const delay = Math.max(0, t - clock);
      clock = t;
      if (event.eventType === 'Delete') state.delete(event.uri);
      else state.set(event.uri, { status: 200, data: event.data ?? null });
      return { delay, message: JSON.stringify([8, 'OnJsonApiEvent', { uri: event.uri, eventType: event.eventType, data: event.data ?? null }]) };
    },
    phase() {
      return state.get(PHASE_URI)?.data ?? null;
    },
  };
}

/** Assets do client não existem fora dele; a LCU simulada redireciona para a CDN do CommunityDragon. */
export function assetRedirect(path) {
  const clean = path.split('?')[0];
  if (clean.includes('..') || !/^[\w./-]+$/.test(clean)) return null;
  const gameData = /^\/lol-game-data\/assets\/(.+)$/.exec(clean);
  if (gameData) return `${CDN}/rcp-be-lol-game-data/global/default/${gameData[1].toLowerCase()}`;
  const fe = /^\/fe\/([\w-]+)\/(.+)$/.exec(clean);
  if (fe) return `${CDN}/rcp-fe-${fe[1]}/global/default/${fe[2].toLowerCase()}`;
  return null;
}
```

```js
// LCU simulada: reproduz uma gravação de docs/dump/sessions por HTTP e WebSocket, para desenvolver no navegador.
// Uso: node tools/mock-lcu/server.mjs [--idle] [--speed 2] [--until ChampSelect] [--session nome]
import { readFileSync } from 'node:fs';
import { createServer } from 'node:http';
import { WebSocketServer } from 'ws';
import { assetRedirect, createReplay } from './replay.mjs';

export const PORT = 47616;

const args = process.argv.slice(2);
const flag = (name) => args.includes(`--${name}`);
const option = (name, fallback) => (args.includes(`--${name}`) ? args[args.indexOf(`--${name}`) + 1] : fallback);

const session = option('session', 'ferramenta-de-treino');
if (!/^[a-z0-9-]+$/.test(session)) {
  console.error('erro: --session aceita só minúsculas, dígitos e hífen');
  process.exit(1);
}
const speed = Math.max(0.1, Number(option('speed', '1')) || 1);
const until = option('until', null);
const idle = flag('idle');

let lines;
try {
  lines = readFileSync(`docs/dump/sessions/${session}.jsonl`, 'utf8').split('\n').filter(Boolean).map((l) => JSON.parse(l));
} catch (e) {
  console.error(`erro: não foi possível ler a gravação "${session}": ${e.message}`);
  process.exit(1);
}
const replay = createReplay(lines, { idle });

const server = createServer((req, res) => {
  const redirect = assetRedirect(req.url);
  if (redirect) return void res.writeHead(302, { Location: redirect }).end();
  if (req.method !== 'GET') return void res.writeHead(405).end();
  const { status, data } = replay.get(req.url);
  res.writeHead(status, { 'Content-Type': 'application/json' }).end(JSON.stringify(data));
});

const wss = new WebSocketServer({ server });
let playing = false;
let paused = false;

function broadcast(message) {
  for (const client of wss.clients) if (client.readyState === 1) client.send(message);
}

function step() {
  if (paused) return;
  const event = replay.next();
  if (!event) return void console.log('fim da gravação');
  setTimeout(() => {
    broadcast(event.message);
    if (until && replay.phase() === until) {
      paused = true;
      console.log(`pausado em ${until}. Enter para continuar.`);
      return;
    }
    step();
  }, event.delay / speed);
}

wss.on('connection', (socket) => {
  socket.on('message', () => {
    // A primeira inscrição dispara a reprodução; todos os clientes compartilham a mesma linha do tempo.
    if (playing || idle) return;
    playing = true;
    step();
  });
});

process.stdin.on('data', () => {
  if (!paused) return;
  paused = false;
  step();
});

server.listen(PORT, '127.0.0.1', () => console.log(`LCU simulada em http://127.0.0.1:${PORT} (${session}${idle ? ', ociosa' : `, velocidade ${speed}x`})`));
```
